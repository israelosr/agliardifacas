  <script>
async function carregarCatalogo() {
    const container = document.getElementById('grid-catalogo');
    
    // Lista de arquivos (No CMS real, você precisará de uma lista ou API)
    // Para simplificar, vamos buscar um arquivo 'index.json' que o CMS pode gerar
    // ou ler uma lista pré-definida.
    
    try {
        // Exemplo: buscando dados de um arquivo centralizado ou API do GitHub
        const response = await fetch('data/produtos.json'); 
        const produtos = await response.json();

        container.innerHTML = ''; // Limpa o carregando

        produtos.forEach(faca => {
            const card = `
                <div class="group bg-brand-charcoal p-4 rounded-xl border border-white/5 hover:border-gold/50 transition-all">
                    <div class="overflow-hidden rounded-lg mb-4">
                        <img src="${faca.image}" alt="${faca.title}" class="w-full h-64 object-cover group-hover:scale-110 transition-duration-500">
                    </div>
                    <h3 class="font-serif text-xl text-white">${faca.title}</h3>
                    <p class="text-gold font-bold mt-2">${faca.price}</p>
                    <a href="${faca.link}" class="block mt-4 text-center border border-gold/30 py-2 rounded-full text-xs uppercase tracking-widest hover:bg-gold hover:text-black transition">
                        Encomendar
                    </a>
                </div>
            `;
            container.innerHTML += card;
        });
    } catch (error) {
        console.error("Erro ao carregar facas:", error);
        container.innerHTML = '<p class="text-white">Em breve, novas peças exclusivas.</p>';
    }
}

document.addEventListener('DOMContentLoaded', carregarCatalogo);
</script>
