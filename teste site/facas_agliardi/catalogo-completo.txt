            <!-- O Grid onde aparecerão todas as facas -->

    <!-- Footer (Copie o do index.html) -->

    <script>
        // Script para carregar os mesmos dados (ou um arquivo json maior se preferir)
        async function carregarCatalogoCompleto() {
            const container = document.getElementById('grid-catalogo-completo');
            try {
                const response = await fetch('./data/produtos.json');
                const produtos = await response.json();
                
                container.innerHTML = produtos.map(faca => `
                    <div class="group bg-brand-charcoal rounded-xl overflow-hidden border border-white/5 hover:border-gold/30 transition-all duration-500">
                        <div class="relative overflow-hidden h-72">
                            <img src="${faca.image}" alt="${faca.title}" class="w-full h-full object-cover group-hover:scale-110 transition-transform duration-700">
                            <div class="absolute top-4 right-4 bg-black/60 backdrop-blur-md px-3 py-1 rounded text-gold text-xs">Artesanal</div>
                        </div>
                        <div class="p-6">
                            <h3 class="font-serif text-2xl text-white mb-2">${faca.title}</h3>
                            <p class="text-gold font-bold text-xl mb-4">${faca.price}</p>
                            <a href="${faca.link}" class="block w-full text-center bg-transparent border border-gold text-gold py-3 rounded-full hover:bg-gold hover:text-black transition-colors font-bold uppercase tracking-widest text-xs">
                                Consultar Disponibilidade
                            </a>
                        </div>
                    </div>
                `).join('');
            } catch (error) {
                console.error("Erro:", error);
            }
        }
        document.addEventListener('DOMContentLoaded', carregarCatalogoCompleto);
    </script>
